# AI Auto-Blog System Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Daily automated pipeline — RSS → Groq (Llama 3.3 70B) → Unsplash → Supabase → IndexNow — publishing one bilingual (EN/FR) immigration blog post per day at zero cost.

**Architecture:** A Netlify scheduled function runs daily at 7am UTC. It fetches three RSS feeds, picks the newest unprocessed item (dedup by source_url), generates a full EN+FR article via Groq, attaches an Unsplash photo, inserts the post into Supabase as published, then pings IndexNow. The blog renders via ISR Next.js pages reading from Supabase.

**Tech Stack:** Netlify Scheduled Functions, Groq SDK (`groq-sdk`), Unsplash API (fetch), `rss-parser`, `marked` + `isomorphic-dompurify` (markdown → sanitized HTML), Supabase (service role client), Next.js ISR, IndexNow (existing `src/lib/indexnow.ts`).

// src/lib/blog/rss.ts
import Parser from "rss-parser";

export type RssItem = {
  title: string;
  link: string;
  contentSnippet: string;
  pubDate: string;
};

const FEEDS = [
  "https://www.canada.ca/en/immigration-refugees-citizenship/news.atom",
  "https://www.cicnews.com/feed/",
  "https://www.canadavisa.com/canada-immigration-news.rss",
];

const parser = new Parser({ timeout: 10000 });

export async function fetchLatestItems(): Promise<RssItem[]> {
  const results = await Promise.allSettled(FEEDS.map((url) => parser.parseURL(url)));

  const items: RssItem[] = [];
  for (const result of results) {
    if (result.status === "fulfilled") {
      for (const item of result.value.items.slice(0, 5)) {
        if (item.title && item.link) {
          items.push({
            title: item.title,
            link: item.link,
            contentSnippet: item.contentSnippet ?? item.title,
            pubDate: item.pubDate ?? new Date().toISOString(),
          });
        }
      }
    }
  }

  return items.sort(
    (a, b) => new Date(b.pubDate).getTime() - new Date(a.pubDate).getTime(),
  );
}
```

// src/lib/blog/generate.ts
import Groq from "groq-sdk";
import type { RssItem } from "./rss";

export type GeneratedArticle = {
  slug: string;
  title_en: string;
  title_fr: string;
  excerpt_en: string;
  excerpt_fr: string;
  content_en: string;
  content_fr: string;
  meta_desc_en: string;
  meta_desc_fr: string;
  unsplash_keyword: string;
};

const groq = new Groq({ apiKey: process.env.GROQ_API_KEY });

const SYSTEM_PROMPT = `You are an expert Canadian immigration content writer.
Generate SEO-optimized bilingual blog posts for an immigration consulting firm (IDLOGICE) based in Montreal.
Always respond with valid JSON only — no markdown fences, no extra text.`;

export async function generateArticle(item: RssItem): Promise<GeneratedArticle> {
  const userPrompt = `
News item to write about:
Title: ${item.title}
Summary: ${item.contentSnippet}
Source: ${item.link}

Generate a blog post JSON with exactly these keys:
- slug: lowercase-hyphenated URL slug, max 60 chars, no dates
- title_en: SEO title in English, 50-60 chars
- title_fr: SEO title in French, 50-60 chars
- excerpt_en: 1-2 sentence summary in English, 120-160 chars
- excerpt_fr: 1-2 sentence summary in French, 120-160 chars
- content_en: Full article in English, 600-900 words, markdown with ## headings
- content_fr: Full article in French, 600-900 words, markdown with ## headings
- meta_desc_en: Meta description in English, 140-155 chars
- meta_desc_fr: Meta description in French, 140-155 chars
- unsplash_keyword: 2-3 word English phrase for image search (e.g. "canada passport")

Rules:
- Include primary keyword near the start of title and first paragraph
- Content must be accurate and helpful for immigrants to Canada
- Tone: professional, warm, helpful
- Do NOT mention IDLOGICE in the content
`;

  const completion = await groq.chat.completions.create({
    model: "llama-3.3-70b-versatile",
    messages: [
      { role: "system", content: SYSTEM_PROMPT },
      { role: "user", content: userPrompt },
    ],
    temperature: 0.7,
    max_tokens: 4000,
    response_format: { type: "json_object" },
  });

  const raw = completion.choices[0]?.message?.content;
  if (!raw) throw new Error("Groq returned empty response");

  const article = JSON.parse(raw) as GeneratedArticle;

  const required: (keyof GeneratedArticle)[] = [
    "slug", "title_en", "title_fr", "excerpt_en", "excerpt_fr",
    "content_en", "content_fr", "meta_desc_en", "meta_desc_fr", "unsplash_keyword",
  ];
  for (const field of required) {
    if (!article[field]) throw new Error(`Groq response missing field: ${field}`);
  }

  return article;
}
```

// src/lib/blog/unsplash.ts
export type UnsplashPhoto = {
  url: string;
  alt: string;
  credit: string;
};

export async function fetchUnsplashPhoto(keyword: string): Promise<UnsplashPhoto | null> {
  const key = process.env.UNSPLASH_ACCESS_KEY;
  if (!key) {
    console.warn("UNSPLASH_ACCESS_KEY not set — skipping image");
    return null;
  }

  const encoded = encodeURIComponent(keyword);
  const res = await fetch(
    `https://api.unsplash.com/search/photos?query=${encoded}&per_page=1&orientation=landscape`,
    { headers: { Authorization: `Client-ID ${key}` } },
  );

  if (!res.ok) {
    console.warn(`Unsplash fetch failed: ${res.status}`);
    return null;
  }

  const data = await res.json();
  const photo = data?.results?.[0];
  if (!photo) return null;

  return {
    url: photo.urls?.regular ?? photo.urls?.full,
    alt: photo.alt_description ?? keyword,
    credit: `Photo by ${photo.user?.name ?? "Unsplash"} on Unsplash`,
  };
}
```

// src/lib/blog/publish.ts
import { createServiceClient } from "../supabase/service";
import { indexNow } from "../indexnow";
import type { GeneratedArticle } from "./generate";
import type { UnsplashPhoto } from "./unsplash";

export async function publishPost(
  article: GeneratedArticle,
  sourceUrl: string,
  sourceTitle: string,
  photo: UnsplashPhoto | null,
): Promise<{ slug: string } | null> {
  const db = createServiceClient();

  const { data, error } = await db
    .from("blog_posts")
    .insert({
      slug: article.slug,
      source_url: sourceUrl,
      source_title: sourceTitle,
      title: { en: article.title_en, fr: article.title_fr },
      excerpt: { en: article.excerpt_en, fr: article.excerpt_fr },
      content: { en: article.content_en, fr: article.content_fr },
      meta_desc: { en: article.meta_desc_en, fr: article.meta_desc_fr },
      image_url: photo?.url ?? null,
      image_alt: photo?.alt ?? null,
      image_credit: photo?.credit ?? null,
      status: "published",
    })
    .select("slug")
    .single();

  if (error) {
    if (error.code === "23505") {
      console.log("Already published:", sourceUrl);
      return null;
    }
    throw new Error(`Supabase insert failed: ${error.message}`);
  }

  const slug = data.slug;

  await indexNow([
    `https://idlogice.com/en/blog/${slug}`,
    `https://idlogice.com/fr/blog/${slug}`,
  ]).catch((e) => console.warn("IndexNow ping failed:", e));

  console.log(`Published: /en/blog/${slug}`);
  return { slug };
}
```

// netlify/functions/blog-generator.mts
import { fetchLatestItems } from "../../src/lib/blog/rss";
import { generateArticle } from "../../src/lib/blog/generate";
import { fetchUnsplashPhoto } from "../../src/lib/blog/unsplash";
import { publishPost } from "../../src/lib/blog/publish";
import { createServiceClient } from "../../src/lib/supabase/service";

export default async function handler() {
  console.log("Blog generator started:", new Date().toISOString());

  const items = await fetchLatestItems();
  if (items.length === 0) {
    console.log("No RSS items found — exiting");
    return;
  }

  const db = createServiceClient();
  let targetItem = null;

  for (const item of items) {
    const { data } = await db
      .from("blog_posts")
      .select("id")
      .eq("source_url", item.link)
      .maybeSingle();

    if (!data) {
      targetItem = item;
      break;
    }
  }

  if (!targetItem) {
    console.log("All items already published — exiting");
    return;
  }

  console.log("Generating article for:", targetItem.title);

  const article = await generateArticle(targetItem);
  const photo = await fetchUnsplashPhoto(article.unsplash_keyword);
  const result = await publishPost(article, targetItem.link, targetItem.title, photo);

  if (result) console.log("Done. Slug:", result.slug);
}

export const config = { schedule: "0 7 * * *" };
```

// src/components/blog/PostBody.tsx
import { marked } from "marked";
import DOMPurify from "isomorphic-dompurify";

type Props = { content: string };

export function PostBody({ content }: Props) {
  const rawHtml = marked.parse(content) as string;
  const safeHtml = DOMPurify.sanitize(rawHtml, {
    ALLOWED_TAGS: [
      "h1","h2","h3","h4","p","ul","ol","li","strong","em","a","blockquote",
      "code","pre","br","hr","img",
    ],
    ALLOWED_ATTR: ["href", "src", "alt", "title", "rel", "target"],
    FORCE_BODY: true,
  });

  return (
    <div
      className="prose prose-lg max-w-none dark:prose-invert prose-headings:font-bold prose-a:text-primary"
      // Content is AI-generated, sanitized with DOMPurify before render
      // eslint-disable-next-line react/no-danger
      dangerouslySetInnerHTML={{ __html: safeHtml }}
    />
  );
}

            <p className="absolute bottom-2 right-3 text-xs text-white/70">{post.image_credit}</p>
          )}
        </div>
      )}
      <p className="text-sm text-muted-foreground mb-3">{date}</p>
      <h1 className="text-4xl font-bold leading-tight mb-8">{title}</h1>
      <PostBody content={content} />
      {post.source_url && (
        <p className="mt-12 text-sm text-muted-foreground border-t pt-4">
          Source:{" "}
          <a
            href={post.source_url}
            target="_blank"
            rel="noopener noreferrer"
            className="underline"
          >
            {post.source_title ?? post.source_url}
          </a>
        </p>
      )}
    </article>
  );
}
```
